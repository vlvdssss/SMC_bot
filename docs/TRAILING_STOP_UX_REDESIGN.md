```python
class TrailingProtectionPanel:
    """
    Современный блок настроек защиты позиций
    """
    
    def __init__(self):
        self.mode = "smart"  # static | smart
        
        # ШАГ 1: BreakEven
        self.be_trigger = "progress"  # progress | profit
        self.be_at_progress = 60  # 0-100%
        self.be_at_profit = 5.0  # $ или pips
        self.be_lock_mode = "partial"  # full (0%) | partial (50%)
        
        # ШАГ 2: Trailing
        self.trail_frequency = "moderate"  # aggressive | moderate | conservative
        self.trail_distance = 15  # пипсов
        
    def render(self):
        # БЛОК 1: Режим
        self.render_mode_selector()
        
        if self.mode == "smart":
            # БЛОК 2: BreakEven
            self.render_breakeven_block()
            
            # БЛОК 3: Trailing
            self.render_trailing_block()
            
            # БЛОК 4: Preview
            self.render_visualization()
    
    def render_mode_selector(self):
        """
        ○ Статичный SL (не двигается)
        ● Умная защита (автоматическое управление)
        """
        pass
    
    def render_breakeven_block(self):
        """
        ШАГ 1: ПЕРЕВОД В БЕЗУБЫТОК
        
        Триггер:
        - ● По прогрессу к цели: [====60%====]
        - ○ По фиксированной прибыли: [__5__] $
        
        Позиция SL:
        - ○ Точно в entry (0%)
        - ● С защитой прибыли (50%)
        """
        if self.be_trigger == "progress":
            # Показываем слайдер 0-100%
            self.render_slider("При достижении % к TP", 0, 100, 60)
        else:
            # Показываем input
            self.render_input("Когда профит ≥", "$", 5.0)
        
        # Позиция SL
        self.render_radio_group([
            ("full", "Точно в entry (без запаса)"),
            ("partial", "С защитой 50% прибыли")
        ])
    
    def render_trailing_block(self):
        """
        ШАГ 2: ТРЕЙЛИНГ

            ("conservative", "10 пипсов", "Дать пространство")
        ])
        
        # Input дистанции

def validate_settings(self):
    """Проверка конфликтов"""
    
    # Если BE триггер > 80%, предупредить
    if self.be_at_progress > 80:
        show_warning("⚠️ Высокий порог может не сработать до TP")
    
    # Если trail_distance слишком мал
    if self.trail_distance < 10:
        show_warning("⚠️ Малая дистанция → частые ложные срабатывания")
    
    # Если preserve = 0% но trail включен
    if self.be_lock_mode == "full" and self.trail_enabled:
        show_info("ℹ️ SL в entry, trailing начнётся сразу после")
```
